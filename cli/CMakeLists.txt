add_executable(wirelens_cli main.cpp)
set_target_properties(wirelens_cli PROPERTIES OUTPUT_NAME wirelens)
add_executable(wirelens ALIAS wirelens_cli)
target_link_libraries(wirelens_cli PRIVATE wirelens_core)
target_compile_features(wirelens_cli PRIVATE cxx_std_20)

endif()
